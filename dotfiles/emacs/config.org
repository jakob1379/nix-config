** Early init
Early startup settings live in =~/.emacs.d/early-init.el= so they are applied before package and frame initialization. Keeping them in this file was too late for package startup and would also make tangling try to rewrite Home Manager's symlinked early-init file.

  "Run FUNCTION after startup and SECONDS idle seconds."
  (add-hook 'emacs-startup-hook
            (list 'lambda nil
                  (list 'run-with-idle-timer seconds nil
                        (list 'quote function)))))

(defun my/forward-sexp-or-string (&optional arg)
  "Like `forward-sexp', but jump out of a string when point is inside one."
  (interactive "^p")
  (let ((ppss (syntax-ppss)))
    (if (nth 3 ppss)
        (goto-char (scan-sexps (nth 8 ppss) 1))
      (forward-sexp arg))))

(defun my/backward-sexp-or-string (&optional arg)
  "Like `backward-sexp', but jump to the opening quote when inside a string."
  (interactive "^p")
  (let ((ppss (syntax-ppss)))
    (if (nth 3 ppss)
        (goto-char (nth 8 ppss))
      (backward-sexp arg))))

(global-set-key (kbd "C-M-<right>") 'my/forward-sexp-or-string)
(global-set-key (kbd "C-M-<left>")  'my/backward-sexp-or-string)

*** Corfu - Completion Overlay Region Function
Corfu is a small CAPF UI. Eglot publishes LSP completions through CAPF, so the efficient path is Eglot -> CAPF -> Corfu, with Cape only for a few explicit fallback completions.
#+BEGIN_SRC emacs-lisp
(use-package emacs
  :straight nil
  :init
  (when (boundp 'treesit-enabled-modes)
    (setq treesit-enabled-modes t))
  (when (boundp 'completion-eager-update)
    (setq completion-eager-update t))
  (when (boundp 'completion-eager-display)
    (setq completion-eager-display 'auto))
  (when (boundp 'minibuffer-visible-completions)
    (setq minibuffer-visible-completions 'up-down))
  (when (boundp 'eldoc-help-at-pt)
    (setq eldoc-help-at-pt t))
  (when (boundp 'kill-region-dwim)
    (setq kill-region-dwim 'emacs-word))
  (when (boundp 'ibuffer-human-readable-size)
    (setq ibuffer-human-readable-size t))
  (when (boundp 'display-fill-column-indicator-warning)
    (setq display-fill-column-indicator-warning nil))
  :custom
  ;; Let TAB indent first and complete when point is already indented.
  (tab-always-indent 'complete)
  ;; Emacs 30+: avoid adding Ispell as a noisy text-mode CAPF.
  (text-mode-ispell-word-completion nil)
  ;; Hide mode-inapplicable commands from M-x.
  (read-extended-command-predicate #'command-completion-default-include-p))

(use-package corfu
  :custom
  (corfu-auto t)
  (corfu-auto-delay 0.2)
  (corfu-auto-prefix 2)
  (corfu-cycle t)
  (corfu-preselect 'prompt)
  (corfu-quit-no-match 'separator)
  (corfu-count 8)
  (corfu-popupinfo-delay '(0.4 . 0.2))
  (corfu-popupinfo-max-height 20)
  :bind (:map corfu-map
              ("C-n" . corfu-next)
              ("C-p" . corfu-previous)
              ("C-g" . corfu-quit)
              ("M-g" . corfu-info-location)
              ("M-h" . corfu-info-documentation)
              ("M-d" . corfu-popupinfo-toggle)
              ("M-p" . corfu-popupinfo-scroll-down)
              ("M-n" . corfu-popupinfo-scroll-up))
  :init
  (global-corfu-mode)
  :config
  (require 'corfu-popupinfo)

*** LaTeX / AUCTeX
AUCTeX registers the =.tex= → =LaTeX-mode= remap itself from its autoloads, so no =:mode= is
needed here. Nothing below loads at startup: every hook and key target is an autoload, so
=tex.el=, =tex-fold.el=, =reftex.el= and =cdlatex.el= arrive with the first =.tex= buffer.

The package is declared as =tex= with =:straight auctex= rather than as =auctex=: =tex.el= is
the feature that actually gets loaded when a =.tex= buffer opens, which is what makes =:config=
run while still deferring ([[https://github.com/jwiegley/use-package/issues/379#issuecomment-246161500][use-package#379]]). =TeX-after-compilation-finished-functions= must live in
=:config= because it is a plain =defvar=, not a =defcustom= — =:custom= silently drops it.

#+BEGIN_SRC emacs-lisp
(use-package tex
  :defer t
  :straight auctex
  :custom
  (TeX-auto-save t)
  (TeX-parse-self t)
  (TeX-save-query nil)                    ;; don't ask before compiling
  (TeX-source-correlate-method 'synctex)
  (TeX-source-correlate-start-server t)   ;; inverse search from the PDF
  (TeX-electric-math nil)                 ;; cdlatex owns $
  (TeX-electric-sub-and-superscript nil)  ;; cdlatex owns _ and ^
  (LaTeX-electric-left-right-brace nil)   ;; electric-pair-mode owns braces
  (TeX-view-program-selection '((output-pdf "PDF Tools")))
  (preview-auto-cache-preamble t)
  ;; Sectioning size. The number is a *cumulative* scale from
  ;; `font-latex-sectioning-5-face' upward, so 1.1 gives
  ;; subsubsection 1.1, subsection 1.21, section 1.331, chapter 1.46, part 1.61.
  ;; Must go through :custom — it has a :set handler, so setq is a no-op.
  ;; Colour comes from the theme, which restyles each level individually.
  (font-latex-fontify-sectioning 1.1)
  ;; \title is in font-latex's "textual" class, which has no size. Promote it
  ;; to the largest sectioning face; the textual class still supplies colour.
  (font-latex-user-keyword-classes
   '(("documenttitle" (("title" "{")) font-latex-sectioning-0-face command)))
  ;; AUCTeX fills prettify-symbols-alist with ~725 entries (\alpha -> α);
  ;; unprettify-at-point keeps the raw macro editable under the cursor.
  (prettify-symbols-unprettify-at-point 'right-edge)
  :config
  ;; plain defvar hook: re-render the PDF in pdf-tools after each compile
  (add-hook 'TeX-after-compilation-finished-functions
            #'TeX-revert-document-buffer)
  :hook ((LaTeX-mode . TeX-source-correlate-mode)
         (LaTeX-mode . TeX-fold-mode)
         (LaTeX-mode . prettify-symbols-mode)
         (LaTeX-mode . turn-on-reftex)))
